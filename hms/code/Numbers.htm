import java.io.*;

public class Numbers implements Serializable
{
	private int cnum;
	private int rnum;
	private int tnum;
	private int rbnum;
	private int tbnum;

	public Numbers(int c, int r, int t, int rb, int tb)
	{
		cnum = c;
		rnum = r;
		tnum = t;
		rbnum = rb;
		tbnum = tb;
	}

	public int getCNum()
	{
		return cnum;
	}

	public int getRNum()
	{
		return rnum;
	}

	public int getTNum()
	{
		return tnum;
	}

	public int getRBNum()
	{
		return rbnum;
	}

	public int getTBNum()
	{
		return tbnum;
	}
}
