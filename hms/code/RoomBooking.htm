import java.io.*;

public class RoomBooking implements Serializable
{
	private String Number;
	private String Customer_Number;
	private String Room_Number;
	private String Arrival_Date_Day;
	private String Arrival_Date_Month;
	private String Arrival_Date_Year;
	private String Departure_Date_Day;
	private String Departure_Date_Month;
	private String Departure_Date_Year;
	private String Standard;
	private String Checked_In;
	private String No_People;
	private String Discount;
	private String Notes;

	public RoomBooking()
	{
		this.Number = "";
		this.Customer_Number = "";
		this.Room_Number = "";
		this.Arrival_Date_Day = "";
		this.Arrival_Date_Month = "";
		this.Arrival_Date_Year = "";
		this.Departure_Date_Day = "";
		this.Departure_Date_Month = "";
		this.Departure_Date_Year = "";
		this.Standard = "";
		this.Checked_In = "";

		this.Notes = "";

		this.Number = num;
		this.Customer_Number = cus;
		this.Room_Number = rom;
		this.Arrival_Date_Day = arra;
		this.Arrival_Date_Month = arrb;
		this.Arrival_Date_Year = arrc;

		this.Standard = stn;
		this.Checked_In = chi;
		this.No_People = nop;
		this.Discount = dis;
		this.Notes = nts;

	}

	public String getCustomerNumber()
	{
		return this.Customer_Number;

	public String getRoomNumber()

	public String getArrivalDateYear()

	public String getDepartureDateDay()
	{
		return this.Departure_Date_Day;
	}

	public String getDepartureDateMonth()
	{
		return this.Departure_Date_Month;
	}

	public String getDepartureDateYear()
	{
		return this.Departure_Date_Year;

	public String getStandard()

		return this.Standard;
	}

	public String getCheckedIn()
	{
		return this.Checked_In;
	}

	public String getNoPeople()
	{
		return this.No_People;
	}

	public String getDiscount()
	{
		return this.Discount;
	}

	public String getNotes()
	{
		return this.Notes;
	}

	public void setNumber(String x)
	{
		this.Number = x;
	}

	public void setCustomerNumber(String x)
	{
		this.Customer_Number = x;
	}

	public void setRoomNumber(String x)
	{
		this.Room_Number = x;
	}

	public void setArrivalDateDay(String x)
	{
		this.Arrival_Date_Day = x;
	}

	public void setArrivalDateMonth(String x)

		this.Arrival_Date_Month = x;

	public void setArrivalDateYear(String x)

		this.Departure_Date_Day = x;

	{
		this.Departure_Date_Month = x;
	}

	public void setDepartureDateYear(String x)
	{
		this.Departure_Date_Year = x;
	}

	{
		this.Standard = x;
